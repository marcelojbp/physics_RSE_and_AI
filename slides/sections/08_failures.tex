\begin{frame}{Plausible is not correct}{$\Delta = 0$: which curve is right?}
  \vspace{1ex}
  \begin{center}
    \begin{tikzpicture}[x=26mm,y=24mm]
      \draw[unibasLightGrey,line width=0.4pt] (0,0) -- (3,0) (0,0) -- (0,1.05);
      \node[lbl,below] at (3,0) {$\Omega t/2\pi$};
      \node[lbl,left] at (0,1) {$1$};
      \node[lbl,left] at (0,0) {$0$};
      \node[lbl,rotate=90,anchor=south] at (-0.15,0.5) {$\Pe$};
      \foreach \x in {1,2,3} \node[lbl,below] at (\x,0) {$\x$};
      \draw[figModel,line width=1pt,domain=0:3,samples=150,smooth]
        plot (\x,{sin(180*\x)^2});
      \node[font=\scriptsize,text=figModel,anchor=west] at (3.02,0.9) {A};
      \onslide<2->{%
        \draw[figTool,line width=1pt,domain=0:3,samples=200,smooth]
          plot (\x,{sin(360*\x)^2});
        \node[font=\scriptsize,text=figTool,anchor=west] at (3.02,0.55) {B};}
    \end{tikzpicture}
  \end{center}
  \begin{itemize}
    \item<3-> Both are smooth, bounded, periodic: both ``look like Rabi oscillations''
    \item<4-> A is $\sin^2(\Omega t/2)$: correct. B is the bug, $\Omega$ instead of $\Omega/2$
    \item<5-> The eye checks shape; the test checks \term{the number}
  \end{itemize}
\end{frame}

\begin{frame}{The bug your tests cannot see}{A sign in the Hamiltonian term}
  \begin{equation*}
    -i[H, \rho] \;\longrightarrow\; +i[H, \rho]
  \end{equation*}

  \begin{itemize}
    \item<2-> Every population is unchanged: Rabi, decay, Ramsey and echo tests all pass
    \item<3-> Only phases flip: at $\Delta = 0$, $\langle\sigma_y\rangle(t) = \sin(\Omega t)$
          becomes $-\sin(\Omega t)$
    \item<4-> Your tests decide which bugs you can see: test every quantity you will publish
  \end{itemize}
\end{frame}

\begin{frame}{You are the author}
  \begin{itemize}
    \item \textbf{Accountability}: the agent is not an author; every claim in the paper is yours
    \item<2-> \textbf{Data}: unpublished data, collaborators' code and personal data
          leave your machine when the agent reads them; check what your institution allows
    \item<3-> \textbf{Disclosure}: journals ask how AI was used; know your journal's policy
    \item<4-> \textbf{Reproducibility}: record which tool and model helped, and when
  \end{itemize}
\end{frame}
