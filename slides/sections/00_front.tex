\begin{frame}{Abundance outpaces verification}{Mathematics in the age of AI}
  \vspace{2ex}
  \begin{center}
    \begin{minipage}{0.78\linewidth}
      \large\rmfamily ``AI-generated proofs will accumulate faster than they can be
      verified''
      \par\vspace{0.8ex}\hfill\normalsize --- T.~Tao, ICM lecture (2026)
    \end{minipage}
  \end{center}
  \vspace{2ex}

  \begin{itemize}
    \item From proof \emph{scarcity} to proof \emph{abundance}
    \item Journals, peer review, careers: all built for scarcity
    \item<2-> Physics is no different: code, numerics and derivations are now cheap
  \end{itemize}

\begin{frame}{The thesis of this afternoon}
  \vspace{4ex}
  \begin{center}
    \ckeybox[0.72\linewidth]{\large
      AI makes code cheap.\\[0.4ex] Verification becomes the bottleneck.}
  \end{center}
  \vspace{3ex}

  \begin{itemize}
    \item<2-> So: invest in verification
    \item<3-> Research software engineering (RSE): the practices that make research
          code trustworthy
    \item<4-> Version control, project structure, tests, automation
    \item<5-> Invented for unreliable humans; they work just as well for agents
  \end{itemize}
\end{frame}
