\begin{frame}{Backup: inside the transformer}{Attention}
  \begin{equation*}
    \mathrm{Attention}(Q, K, V) = \mathrm{softmax}\!\left(\frac{Q K^{\mathsf{T}}}{\sqrt{d}}\right) V
  \end{equation*}

  \begin{itemize}
    \item $Q$, $K$, $V$: queries, keys and values, linear maps of the token vectors
    \item $d$: dimension of the keys; softmax acts row by row
    \item Each token gathers information from all earlier tokens in the context
    \item Cost grows with the square of the context length
  \end{itemize}

  \slidecite{A.~Vaswani et al.\ \emph{Attention is all you need}, Advances in Neural
    Information Processing Systems \textbf{30} (2017)}
\end{frame}
