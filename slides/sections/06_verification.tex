\begin{frame}[fragile]{Convergence}{Is the number converged, and is the method what it claims?}
  \begin{equation*}
    \varepsilon(\delta t) \simeq C\,\delta t^{\,p}
    \quad\Longrightarrow\quad
    \frac{\varepsilon(\delta t)}{\varepsilon(\delta t/2)} \simeq 2^{p}
  \end{equation*}
\begin{julia}
ρT(dt) = last(lindblad(H, ρ0, [T]; T1, T2, dt))           # fixed-step RK4
err(dt) = abs(excited_population(ρT(dt)) - exp(-T / T1))  # free decay from |e⟩
@test log2(err(0.02) / err(0.01)) ≈ 4 atol = 0.2          # RK4: p = 4
\end{julia}
  \begin{itemize}
    \item $\varepsilon$: error at fixed time $T$; $\delta t$: step; $p$: order of the method
    \item<2-> Catches a wrong integrator, a missing term, an unconverged result
    \item<3-> Same idea for any cutoff: grid size, Fock space, number of samples
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Two independent routes}
\begin{julia}
using QuantumOptics
b = SpinBasis(1//2)                                     # |e⟩ = spin up
H = Δ/2*sigmaz(b) + Ω/2*sigmax(b)
c_ops = [sqrt(1/T1)*sigmam(b), sqrt(1/(2Tφ))*sigmaz(b)]
_, ρt = timeevolution.master(ts, dm(spindown(b)), H, c_ops)
@test [real(expect(projector(spinup(b)), ρ)) for ρ in ρt] ≈ my_Pe  rtol = 1e-6
\end{julia}
  \begin{itemize}
    \item<2-> Different method, different author: their errors are unlikely to coincide
    \item<3-> Disagreement does not tell you who is wrong; agreement is strong evidence
    \item<4-> Also for derivations: a numerical check of every analytic result
  \end{itemize}
\end{frame}

\begin{frame}{Injection--recovery}{The experimentalist's test}
  \vspace{1ex}
  \begin{center}
    \begin{tikzpicture}[node distance=7mm]
      \node[box=figHuman] (p) {\textbf{Choose}\\ $\theta^\ast$};
      \node[box=figTool,right=of p] (s) {\textbf{Simulate}\\ data + noise};
      \node[box=figModel,right=of s] (f) {\textbf{Fit}\\ your pipeline};
      \node[box=figHuman,right=of f] (c) {\textbf{Compare}\\ $\hat\theta \pm \sigma$ vs $\theta^\ast$};
      \foreach \a/\b in {p/s,s/f,f/c} \draw[flow] (\a) -- (\b);
    \end{tikzpicture}
  \end{center}
  \begin{equation*}
    \left|\hat\theta - \theta^\ast\right| < 3\sigma
  \end{equation*}

  \begin{itemize}
    \item $\theta^\ast$: true parameters; $\hat\theta$: fitted values; $\sigma$: reported errors
    \item<2-> Tests the whole pipeline: model, noise, fit and error bars
    \item<3-> Over many seeds, $(\hat\theta - \theta^\ast)/\sigma$ must look like a unit
          Gaussian: are the error bars honest?
  \end{itemize}
\end{frame}

\begin{frame}{Who checks the tests?}
  \begin{itemize}
    \item An agent writing the code and its tests grades its own homework
    \item<2-> Tests copied from the implementation check nothing:
          test against \emph{physics}, not against the code
    \item<3-> Break the code on purpose: every important test must fail
    \item<4-> You read every line of \cmd{test/}; it is short
  \end{itemize}

  \vspace{3ex}
  \onslide<5->{\centering\ckeybox[0.7\linewidth]{The tests are the part you cannot delegate}\par}
\end{frame}
