\begin{frame}{Stage 2: decoherence}{20 minutes}
  \begin{equation*}
    \frac{\dd\rho}{\dd t} = -i[H, \rho] + \Gamma_1 \mathcal{D}[\sigma_-]\rho
      + \frac{\Gamma_\phi}{2}\mathcal{D}[\sigma_z]\rho
  \end{equation*}
  {\scriptsize\color{unibasGrey}
   $\mathcal{D}[L]\rho = L\rho L^\dagger - \tfrac12\{L^\dagger L, \rho\}$;\;
   $\sigma_- = \ket{g}\bra{e}$;\;
   $\Gamma_1 = 1/T_1$, $\Gamma_\phi = 1/T_\phi$;\;
   $1/T_2 = 1/(2T_1) + 1/T_\phi$}

  \vspace{1ex}
  \textbf{Tests}
  \begin{itemize}
    \item<2-> At all times: $\tr\rho = 1$, $\rho = \rho^\dagger$, eigenvalues $\geq 0$
    \item<3-> $\Omega = 0$, start in $\ket{e}$: $\Pe(t) = \ee^{-t/T_1}$
    \item<4-> $\Omega = \Delta = 0$, start in $\ket{+}$: $|\rho_{eg}(t)| = \tfrac12\ee^{-t/T_2}$
    \item<5-> $T_2 > 2T_1$ is rejected with an error; $T_1, T_2 \to \infty$ gives stage 1
  \end{itemize}
  {\scriptsize\color{unibasGrey}\onslide<4->{$\ket{+} = (\ket{e} + \ket{g})/\sqrt{2}$;
    $\rho_{eg} = \bra{e}\rho\ket{g}$}}
\end{frame}

% ------------------------------------------------------------------------------
\begin{frame}{Stage 3: Ramsey and spin echo}{15 minutes}
  \begin{equation*}
    \Pe(\tau) = \tfrac12\left[1 + \ee^{-\tau/T_2}\cos(\Delta\tau)\right]
  \end{equation*}
  {\scriptsize\color{unibasGrey}Ramsey: $\pi/2$ pulse, free evolution for a time $\tau$,
   $\pi/2$ pulse; ideal pulses $R(\theta) = \ee^{-i\theta\sigma_x/2}$; start in $\ket{g}$;
   $T_1 \to \infty$}

  \vspace{1ex}
  \textbf{Tests}
  \begin{itemize}
    \item<2-> Fringes oscillate at angular frequency $\Delta$, under the envelope $\ee^{-\tau/T_2}$
    \item<3-> Echo, $\pi/2$--$\tau/2$--$\pi$--$\tau/2$--$\pi/2$: result independent of
          a static detuning $\delta$
    \item<4-> Average over random static $\delta$: Ramsey decays, echo does not
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------------------
\begin{frame}{Stage 4: a synthetic experiment}{15 minutes}
  \begin{equation*}
    P(\tau) = B + A\,\ee^{-\tau/T_2}\cos(\Delta\tau + \varphi)
  \end{equation*}
  {\scriptsize\color{unibasGrey}Fit model for Ramsey data; $A$, $B$, $\varphi$: amplitude,
   offset, phase. Data: stage 3 with $N$ measurements per point, binomial noise}

  \vspace{1ex}
  \textbf{Tests}
  \begin{itemize}
    \item<2-> Injection--recovery: fitted $T_2$ and $\Delta$ within $3\sigma$ of the true
          values, fixed seed
    \item<3-> Four times more measurements: error bars about half as large
    \item<4-> Script: save the fit with \cmd{produce\_or\_load}; the file carries the commit
  \end{itemize}
\end{frame}
