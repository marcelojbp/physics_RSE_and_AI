\begin{frame}[fragile]{One layout for every project}{DrWatson}

|-- src/            # the physics: functions, no hard-coded parameters

|-- test/           # runtests.jl: the physics checks

|   |-- exp_raw/    # measurements: read-only, never modified
|   |-- exp_pro/    # processed data

\begin{frame}[fragile]{\texttt{src} computes, scripts orchestrate}
\begin{julia}[title={\hfill\tiny\color{unibasGrey}\smallcapsline{scripts/rabi.jl}}]
using DrWatson
@quickactivate "DrivenSpin"          # this project's environment, from any folder
using DrivenSpin                     # the module in src/

for c in dict_list(Dict(:Δ => [0.0, 2π], :Ω => 2π))   # rad/μs; one run per Δ
    produce_or_load(c, datadir("sims", "rabi")) do c
        H = hamiltonian(c[:Δ], c[:Ω])
        t = range(0, 3, length = 301)                  # μs
        Dict("t" => t, "Pe" => excited_population.(Ref(H), t))
    end
end
\end{julia}
  \begin{itemize}
    \item<2-> Scripts contain calls and parameters, no physics
    \item<3-> The same function serves the script, the test and the notebook
    \item<4-> \cmd{produce\_or\_load}: compute and save, or load if the file exists
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{A test is a claim you can run}
\begin{julia}[title={\hfill\tiny\color{unibasGrey}\smallcapsline{test/runtests.jl}}]
using DrivenSpin, Test

@testset "Rabi" begin
    Δ, Ω = 0.3, 2π                                  # rad/μs
    ΩR = sqrt(Ω^2 + Δ^2)
    H = hamiltonian(Δ, Ω)
    for t in range(0, 3, length = 31)               # μs
        @test excited_population(H, t) ≈ Ω^2 / ΩR^2 * sin(ΩR * t / 2)^2 atol = 1e-10
    end
    @test excited_population(hamiltonian(0.0, Ω), π / Ω) ≈ 1    # π-pulse
end
\end{julia}
  \begin{itemize}
    \item<2-> One \cmd{@testset} per physical claim; its name says which
    \item<3-> \cmd{a $\approx$ b} compares with a \emph{relative} tolerance: near zero, give
          \cmd{atol}
  \end{itemize}
\end{frame}

    \item<2-> Run them before every commit; the agent can run them too
    \item<3-> Keep them fast: under a minute, or nobody runs them
    \item<4-> A failing test is information, not an obstacle
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------------------
\begin{frame}{What makes a good test}
  \begin{itemize}
    \item \textbf{Physics, not implementation}: a formula, a limit, a conservation law
    \item<2-> \textbf{Deterministic}: fix the random seed
    \item<3-> \textbf{Fast}: small systems, short times
    \item<4-> \textbf{It fails when the physics is wrong}: break the code on purpose
          and check
  \end{itemize}

  \vspace{3ex}
  \onslide<5->{\centering\ckeybox[0.7\linewidth]{A test that cannot fail checks nothing}\par}
\end{frame}

\begin{frame}[fragile]{The whole configuration}
\begin{terminal}[title={\hfill\tiny\color{unibasGrey}\smallcapsline{.github/workflows/CI.yml}}]
name: CI
on: [push, pull_request]
jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: julia-actions/setup-julia@v2
      - uses: julia-actions/cache@v2
      - uses: julia-actions/julia-buildpkg@v1
      - uses: julia-actions/julia-runtest@v1
\end{terminal}
  \begin{itemize}
    \item<2-> Copy it once; it works for any Julia project
    \item<3-> Results under the \emph{Actions} tab of the repository on GitHub
  \end{itemize}
\end{frame}
